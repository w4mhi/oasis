# OASIS — Python dependencies
# Single source of truth. Installed by: python3 scripts/setup-server.py
#
# All other imports across the project are Python stdlib — no additional
# packages required.
#
# System tools (ogr2ogr, tippecanoe) are in requirements-system.txt.

# Versions are pinned to exactly what is vendored in server/wheels/ so the
# offline resolver and the documented set can never drift apart. Bump these
# together with the wheels (see scripts/create-oasis-offline.py).

# Web framework — server/app.py, services/graywolf/enable-graywolf-api.py
Flask==3.1.3

# Compiled transitive dep of Jinja2/Flask — vendored per platform/python.
MarkupSafe==3.0.3

# Production WSGI server — replaces Flask dev server on the Pi.
# Both 23.0.0 and 26.0.0 are vendored: pip selects 23.0.0 on Python 3.9
# (Pi OS Bullseye — 26.x requires >=3.10) and 26.0.0 on 3.10+ automatically
# via each release's Requires-Python. A range (not env markers) is used because
# `pip download --python-version` evaluates markers against the host, not the
# target interpreter. (gunicorn is POSIX-only and is skipped on Windows.)
gunicorn>=23.0.0,<27

# System stats (CPU, RAM, disk, uptime) — services/graywolf/enable-graywolf-api.py
psutil==7.2.2

# Windows console ANSI support for click (Flask's CLI dependency). Pinned
# unconditionally — without a marker — because `pip download` evaluates
# click's `platform_system == "Windows"` marker against the build host, so a
# non-Windows build silently drops it and the Windows dev server then crashes
# with ModuleNotFoundError: colorama. Pure none-any wheel; harmless no-op on
# Linux/macOS (imported only on win32).
colorama==0.4.6

# Satellite pass prediction (services/satellites/predict.py). Pure-Python SGP4
# propagator + Skyfield's astronomy layer; numpy is Skyfield's compiled dep
# (aarch64 wheels are prebuilt on PyPI — no per-arch compile). No .bsp/ephemeris
# is used; timescale is built with builtin=True so nothing is fetched at runtime.
skyfield>=1.49
sgp4>=2.23
numpy>=1.26
